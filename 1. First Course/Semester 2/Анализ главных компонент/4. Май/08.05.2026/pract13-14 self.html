🔍 Практика 13: Сравнительный анализ методов + Выбор оптимального

Цель: Научиться обоснованно выбирать метод многомерного анализа на основе характеристик данных и требований к интерпретации.

📌 Расшифровка сокращений (повторение + новые)

PCA – Principal Component Analysis (линейное сжатие по дисперсии)
FA – Factor Analysis (латентные факторы)
CA – Correspondence Analysis (контингентные таблицы)
MCA – Multiple Correspondence Analysis (категориальные данные)
MDS – Multidimensional Scaling (расстояния)
Isomap – изометрическое отображение (геодезические расстояния)
KMO – Kaiser-Meyer-Olkin (мера пригодности FA)
EVR – Explained Variance Ratio (доля дисперсии)

1. Диагностика данных (автоматический выбор метода)

import pandas as pd
import numpy as np
from sklearn.decomposition import PCA, FactorAnalysis, KernelPCA
from sklearn.manifold import Isomap
from sklearn.preprocessing import StandardScaler
import prince

def diagnose_data(df):
    """Автоматическая диагностика и рекомендация метода"""
    n, p = df.shape
    numeric_cols = df.select_dtypes(include=[np.number]).columns
    cat_cols = df.select_dtypes(include=['object', 'category']).columns
    
    # KMO тест (для FA) – упрощённо
    kmo_value = 0.0
    if len(numeric_cols) > 1:
        corr = df[numeric_cols].corr().values
        kmo_value = np.abs(corr).mean()  # proxy
    
    # Простая проверка нелинейности (ранговая корреляция расстояний)
    linearity_score = 1.0  # заглушка
    
    if len(cat_cols) > len(numeric_cols):
        return "MCA (анализ категориальных данных)"
    elif kmo_value > 0.7 and linearity_score > 0.8:
        return "FA (факторный анализ)"
    elif linearity_score < 0.5:
        return "Isomap / KernelPCA (нелинейное)"
    else:
        return "PCA (анализ главных компонент)"
\[ \text{KMO} = \frac{\sum\sum_{i\neq j} r_{ij}^2}{\sum\sum_{i\neq j} r_{ij}^2 + \sum\sum_{i\neq j} p_{ij}^2} \] (где \(r_{ij}\) – парные корреляции, \(p_{ij}\) – частные)

⚙️ Практика 14: Production Pipeline + Case Study

Цель: Создать автоматизированный пайплайн для внедрения методов многомерного анализа в промышленную среду.

1. Универсальный класс снижения размерности

from sklearn.pipeline import Pipeline
from sklearn.preprocessing import StandardScaler
from sklearn.decomposition import PCA, FactorAnalysis, KernelPCA
from sklearn.manifold import Isomap
import prince

class UniversalDimReduction:
    def __init__(self, method='auto', n_components=2):
        self.method = method
        self.n_components = n_components
        self.model = None

    def fit(self, X, y=None):
        if self.method == 'auto':
            self.method = self._choose_method(X)
        if self.method == 'pca':
            self.model = Pipeline([('scaler', StandardScaler()), ('pca', PCA(n_components=self.n_components))])
        elif self.method == 'fa':
            self.model = Pipeline([('scaler', StandardScaler()), ('fa', FactorAnalysis(n_components=self.n_components))])
        elif self.method == 'mca':
            self.model = prince.MCA(n_components=self.n_components)
        elif self.method == 'isomap':
            self.model = Isomap(n_components=self.n_components)
        self.model.fit(X)
        return self

    def transform(self, X):
        return self.model.transform(X)

    def _choose_method(self, X):
        # упрощённый эвристический выбор
        if X.select_dtypes(include=['object', 'category']).shape[1] > 0:
            return 'mca'
        else:
            return 'pca'

2. Case Study: Retail Analytics (сегментация клиентов)

import pandas as pd
from sklearn.cluster import KMeans
import prince
from sklearn.decomposition import PCA

# Загрузка данных (числовые + категориальные)
df = pd.read_csv('retail.csv')
num_cols = ['total_spend', 'num_items', 'time_in_store']
cat_cols = ['category', 'day_of_week', 'payment_method']

# PCA для числовых
scaler = StandardScaler()
X_num = scaler.fit_transform(df[num_cols])
pca = PCA(n_components=3)
X_pca = pca.fit_transform(X_num)

# MCA для категориальных
mca = prince.MCA(n_components=3)
X_mca = mca.fit_transform(df[cat_cols]).values

# Объединение и кластеризация
X_combined = np.hstack([X_pca, X_mca])
kmeans = KMeans(n_clusters=5, random_state=42)
clusters = kmeans.fit_predict(X_combined)

df['cluster'] = clusters
profile = df.groupby('cluster').agg({
    'total_spend': 'mean',
    'num_items': 'mean',
    'category': lambda x: x.mode()[0],
    'payment_method': lambda x: x.mode()[0]
})
print(profile)

3. Deployment (Flask API + Streamlit)

# Flask API
from flask import Flask, request, jsonify
app = Flask(__name__)

@app.route('/predict', methods=['POST'])
def predict():
    data = request.json
    X = pd.DataFrame(data)
    reducer = UniversalDimReduction(method='auto')
    reducer.fit(X)
    X_red = reducer.transform(X)
    kmeans = KMeans(n_clusters=5).fit(X_red)
    clusters = kmeans.predict(X_red)
    return jsonify({'clusters': clusters.tolist()})

if __name__ == '__main__':
    app.run(debug=True)
