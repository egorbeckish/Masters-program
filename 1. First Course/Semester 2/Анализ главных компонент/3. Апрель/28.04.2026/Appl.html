Бэкенд на FastAPI + scikit-learn / prince

# main.py
from fastapi import FastAPI, HTTPException
from pydantic import BaseModel
import numpy as np
import pandas as pd
from sklearn.decomposition import PCA
from sklearn.preprocessing import StandardScaler
from sklearn.cluster import KMeans
import prince

app = FastAPI(title="Analytics API")

class DataInput(BaseModel):
    data: list  # список списков (строки)
    method: str  # "pca", "mca", "kmeans"
    n_components: int = 2

# Глобальный кэш моделей (можно сохранять в Redis)
models = {}

@app.post("/transform")
def transform(input: DataInput):
    df = pd.DataFrame(input.data)
    method = input.method
    n = input.n_components
    key = f"{method}_{n}"
    if key not in models:
        if method == "pca":
            scaler = StandardScaler()
            X_scaled = scaler.fit_transform(df)
            model = PCA(n_components=n)
            X_proj = model.fit_transform(X_scaled)
            models[key] = (scaler, model)
        elif method == "mca":
            model = prince.MCA(n_components=n, one_hot=True)
            X_proj = model.fit_transform(df).values
            models[key] = model
        elif method == "kmeans":
            scaler = StandardScaler()
            X_scaled = scaler.fit_transform(df)
            model = KMeans(n_clusters=n, random_state=42)
            labels = model.fit_predict(X_scaled)
            X_proj = X_scaled[:, :2]  # для визуализации используем первые 2 компоненты
            models[key] = (scaler, model)
        else:
            raise HTTPException(status_code=400, detail="Unknown method")
    else:
        # Применение существующей модели
        if method == "pca":
            scaler, model = models[key]
            X_scaled = scaler.transform(df)
            X_proj = model.transform(X_scaled)
        elif method == "mca":
            model = models[key]
            X_proj = model.transform(df).values
        elif method == "kmeans":
            scaler, model = models[key]
            X_scaled = scaler.transform(df)
            labels = model.predict(X_scaled)
            X_proj = X_scaled[:, :2]  # упрощённо
    return {"projection": X_proj.tolist(), "explained_variance": getattr(model, "explained_variance_ratio_", None)}

Веб-клиент (React + Chart.js)

// App.js
import React, { useState } from 'react';
import { Scatter } from 'react-chartjs-2';
import axios from 'axios';

function App() {
  const [data, setData] = useState([]);
  const [projection, setProjection] = useState([]);
  const [method, setMethod] = useState('pca');

  const handleFileUpload = (event) => {
    const file = event.target.files[0];
    const reader = new FileReader();
    reader.onload = (e) => {
      const rows = e.target.result.split('\n').map(row => row.split(',').map(Number));
      setData(rows);
    };
    reader.readAsText(file);
  };

  const analyze = async () => {
    const response = await axios.post('http://localhost:8000/transform', {
      data: data,
      method: method,
      n_components: 2
    });
    setProjection(response.data.projection);
  };

  const chartData = {
    datasets: [{
      label: 'Projected points',
      data: projection.map(p => ({ x: p[0], y: p[1] })),
      backgroundColor: 'rgba(230,126,34,0.6)'
    }]
  };

  return (
    <div>
      <input type="file" onChange={handleFileUpload} />
      <select value={method} onChange={e => setMethod(e.target.value)}>
        <option value="pca">PCA</option>
        <option value="mca">MCA</option>
        <option value="kmeans">KMeans</option>
      </select>
      <button onClick={analyze}>Analyze</button>
      <Scatter data={chartData} />
    </div>
  );
}
export default App;

Стратегии дообучения моделей

1. Инкрементальное обновление PCA
Использование sklearn.decomposition.IncrementalPCA для добавления новых данных без пересчёта всей матрицы.
ipca = IncrementalPCA(n_components=2, batch_size=100)
2. Онлайн-обучение K‑means
Алгоритм «MiniBatchKMeans» обновляет кластеры порциями.
mbk = MiniBatchKMeans(n_clusters=3, batch_size=50)
3. Дообучение FA / MCA
Полный пересчёт при накоплении достаточного количества новых наблюдений (например, раз в неделю).
Использовать фоновые задачи (Celery, Airflow).
4. Active Learning
Пользователь помечает неверные кластеры → модель дообучается на этих примерах (алгоритм «learning with drift»).

🔄 Автоматизация дообучения: сохраняйте контрольные суммы данных; при изменении распределения (PSI > 0.1) запускайте переобучение.

🧠 Дообучение в production: checklist

  • ✅ Сохраняйте версии моделей (MLflow, DVC).
  • ✅ Логируйте запросы пользователей (аномально много новых данных – сигнал для переобучения).
  • ✅ Используйте A/B тестирование: старая модель vs новая для 5% трафика.
  • ✅ Внедрите Shadow mode: новая модель работает параллельно, но не влияет на ответ.
  • ✅ Автоматический перезапуск конвейера по расписанию (GitHub Actions, Airflow).
